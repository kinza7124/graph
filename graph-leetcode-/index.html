Number of Provinces

LeetCode 547
There are n cities. Some are connected, some are not. A province is a group of directly or indirectly connected cities. Given an n x n matrix isConnected where isConnected[i][j] = 1 if connected, return total provinces.
Time ComplexityO(V²)
Space ComplexityO(V)

Logic Breakdown

1. Represent: Convert adjacency matrix to adjacency list for easier traversal.
2. Explore: Loop through each node. If not visited, start a DFS/BFS and increment the province count.
// DFS & BFS Solutions Included
class Solution {
public:
    // DFS Approach
    void dfs(int i, vector<vector<int>> &adjList, vector<bool> &visited) {
        visited[i] = true;
        for (auto it: adjList[i]) {
            if(!visited[it]) dfs(it,adjList,visited);
        }
    }

    int findCircleNum(vector<vector<int>>& isConnected) {
        int V = isConnected.size();
        vector<vector<int>> adjList(V);
        for(int i=0;i<V;i++){
            for(int j=0;j<V;j++) {
                if(isConnected[i][j]==1 && i!=j) {
                    adjList[i].push_back(j);
                    adjList[j].push_back(i);
                }
            }
        }
        vector<bool> visited(V, false);
        int count = 0;
        for(int i=0;i<V;i++) {
            if(!visited[i]) {
                count++;
                dfs(i,adjList,visited);
            }
        }
        return count;
    }
};

Number of Islands

LeetCode 200
Given an m x n 2D binary grid which represents a map of '1's (land) and '0's (water), return the number of islands. An island is surrounded by water and formed by connecting adjacent lands horizontally or vertically.
Time ComplexityO(M × N)
Space ComplexityO(M × N)

Logic Breakdown

1. Scan: Iterate through every cell. If you find land ('1') that isn't visited, you found a new island.
2. Flood: Use BFS to mark all connected land cells as visited so they aren't counted twice.
class Solution {
public:
    void bfs(int m,int n, int i,int j,vector<vector<char>>& grid,vector<vector<bool>>& visited) {
        queue<pair<int,int>> q;
        visited[i][j] = true;
        q.push({i,j});
        int dr[4] = {-1, 1, 0, 0};
        int dc[4] = {0, 0, -1, 1};
        while(!q.empty()) {
            int row = q.front().first;
            int col = q.front().second;
            q.pop();
            for(int k=0;k<4;k++) {
                int nrow = row + dr[k], ncol = col + dc[k];
                if(nrow>=0 && ncol>=0 && nrow<m && ncol<n && grid[nrow][ncol]=='1' && !visited[nrow][ncol]) {
                    q.push({nrow,ncol});
                    visited[nrow][ncol] = true;
                }
            }
        }
    }
};

Rotten Oranges

LeetCode 994
Given a grid where 0=empty, 1=fresh, 2=rotten. Every minute, fresh oranges adjacent to rotten ones rot. Return min time for all to rot. If impossible, return -1.
Time ComplexityO(M × N)
Space ComplexityO(M × N)
class Solution {
public:
    int orangesRotting(vector<vector<int>>& grid) {
        int m=grid.size(); int n = grid[0].size();
        queue<pair<pair<int,int>,int>> q;
        vector<vector<int>> visited(m,vector<int>(n,0));
        for(int i=0;i<m;i++) {
            for(int j=0;j<n;j++) {
                if(grid[i][j]==2) { q.push({{i,j},0}); visited[i][j]=2; }
                else if(grid[i][j]==1) visited[i][j]=1;
            }
        }
        int time=0;
        int dr[]={-1,1,0,0}, dc[]={0,0,-1,1};
        while(!q.empty()){
            int r=q.front().first.first; int c=q.front().first.second; int t=q.front().second;
            q.pop(); time=max(t,time);
            for(int k=0;k<4;k++){
                int nr=r+dr[k], nc=c+dc[k];
                if(nr>=0 && nr<m && nc>=0 && nc<n && visited[nr][nc]==1 && grid[nr][nc]==1){
                    q.push({{nr,nc},t+1}); visited[nr][nc]=2;
                }
            }
        }
        for(int i=0;i<m;i++) for(int j=0;j<n;j++) if(visited[i][j]==1) return -1;
        return time;
    }
};

Bipartite Graph

LeetCode 785
A graph is bipartite if nodes can be partitioned into two independent sets such that every edge connects a node in set A to a node in set B. (No two adjacent nodes have the same color).
Time ComplexityO(V + E)
Space ComplexityO(V)
class Solution {
public:
    bool dfs(int node,vector<vector<int>> &adj,vector<int> &color,int c) {
        color[node] = c;
        for(auto it: adj[node]) {
            if(color[it]==-1) { if(!dfs(it,adj,color,!c)) return false; }
            else if(color[it]==c) return false;
        }
        return true;
    }
    bool isBipartite(vector<vector<int>>& graph) {
        int n=graph.size();
        vector<int> color(n,-1);
        for(int i=0;i<n;i++) {
            if(color[i]==-1) { if(!dfs(i,graph,color,0)) return false; }
        }
        return true;
    }
};

Eventual Safe Nodes

LeetCode 802
A node is safe if every possible path starting from that node leads to a terminal node. Terminal nodes have no outgoing edges.
Time ComplexityO(V + E)
Space ComplexityO(V)
class Solution {
public:
    bool dfs(int node,vector<vector<int>> &adj,vector<int> &vis, vector<int> &pVis, vector<int> &check) {
        vis[node]=1; pVis[node]=1;
        for(auto it: adj[node]) {
            if(!vis[it]) { if(dfs(it,adj,vis,pVis,check)) return true; }
            else if(pVis[it]) return true;
        }
        check[node]=1; pVis[node]=0;
        return false;
    }
    vector<int> eventualSafeNodes(vector<vector<int>>& graph) {
        int V = graph.size();
        vector<int> vis(V,0), pVis(V,0), check(V,0), safe;
        for(int i=0;i<V;i++) if(!vis[i]) dfs(i,graph,vis,pVis,check);
        for(int i=0;i<V;i++) if(check[i]) safe.push_back(i);
        return safe;
    }
};

Network Delay Time

LeetCode 743
You are given n nodes and times[i] = (ui, vi, wi). Return the minimum time it takes for all nodes to receive a signal sent from node k. If impossible, return -1. (Dijkstra's Algorithm).
Time ComplexityO(E log V)
Space ComplexityO(V + E)
class Solution {
public:
    int networkDelayTime(vector<vector<int>>& times, int n, int k) {
        vector<vector<pair<int,int>>> adj(n+1);
        for(auto it:times) adj[it[0]].push_back({it[1],it[2]});
        priority_queue<pair<int,int>, vector<pair<int,int>>, greater<pair<int,int>>> pq;
        vector<int> dist(n+1, 1e9);
        dist[k]=0; pq.push({0,k});
        while(!pq.empty()){
            int d = pq.top().first; int node = pq.top().second; pq.pop();
            for(auto it: adj[node]){
                if(d + it.second < dist[it.first]){
                    dist[it.first] = d + it.second; pq.push({dist[it.first], it.first});
                }
            }
        }
        int ans=0;
        for(int i=1;i<=n;i++) {
            if(dist[i]==1e9) return -1;
            ans=max(ans, dist[i]);
        }
        return ans;
    }
};

Cheapest Flights Within K Stops

LeetCode 787 | Constrained Dijkstra
Find the cheapest price from src to dst with at most K stops. We must track stop count in our priority queue pairs.
TimeO(K * E)
SpaceO(V * K)
class Solution {
public:
    int findCheapestPrice(int n, vector<vector<int>>& flights, int src, int dst, int k) {
        vector<vector<pair<int,int>>> adj(n);
        for(auto& f : flights) adj[f[0]].push_back({f[1], f[2]});

        // Pair: {cost, node, stops}
        priority_queue<tuple<int,int,int>, vector<tuple<int,int,int>>, greater<>> pq;
        pq.push({0, src, 0});

        vector<vector<int>> dist(n, vector<int>(k + 2, INT_MAX));
        dist[src][0] = 0;

        while(!pq.empty()) {
            auto [cost, u, stops] = pq.top(); pq.pop();
            if(u == dst) return cost;
            if(stops > k) continue;

            for(auto& [v, w] : adj[u]) {
                if(cost + w < dist[v][stops + 1]) {
                    dist[v][stops + 1] = cost + w;
                    pq.push({cost + w, v, stops + 1});
                }
            }
        }
        return -1;
    }
};

Word Ladder

LeetCode 127 | BFS Shortest Path
A transformation sequence from beginWord to endWord is a sequence such that:
  • Every adjacent pair of words differs by exactly one letter.
  • Every word in the sequence must be in wordList.
Given two words and a dictionary, return the number of words in the shortest transformation sequence. Return 0 if none exist.
TimeO(N * M * 26)
SpaceO(N * M)

Logic Breakdown

1. BFS Selection: Since we need the shortest path in an unweighted scenario (each word change costs 1), BFS is optimal.
2. Dictionary Check: Use an unordered_set for O(1) lookups to see if a generated word exists in the list.
3. Transformation: For the current word, try changing each character from 'a' to 'z'. If the new word is in the set, add it to the queue and remove it from the set (to mark as visited).
class Solution {
public:
    int ladderLength(string beginWord, string endWord, vector<string>& wordList) {
        // Queue stores {currentWord, steps}
        queue<pair<string, int>> q;
        q.push({beginWord, 1});

        // Hash set for O(1) word discovery
        unordered_set<string> st(wordList.begin(), wordList.end());
        st.erase(beginWord);

        while(!q.empty()) {
            string word = q.front().first;
            int steps = q.front().second;
            q.pop();

            if(word == endWord) return steps;

            for(int i = 0; i < word.size(); i++) {
                char original = word[i];
                for(char ch = 'a'; ch <= 'z'; ch++) {
                    word[i] = ch;
                    // If transformed word exists in dictionary
                    if(st.find(word) != st.end()) {
                        st.erase(word); // Mark as visited
                        q.push({word, steps + 1});
                    }
                }
                word[i] = original; // Backtrack for next character position
            }
        }
        return 0;
    }
};

Number of Ways to Arrive at Destination

LeetCode 1976 | Dijkstra + Dynamic Programming
You are in a city with n intersections and bi-directional roads. You want to find the number of ways you can travel from intersection 0 to n-1 in the shortest amount of time. Return the result modulo 109 + 7.
Time ComplexityO(E log V)
Space ComplexityO(V + E)

Logic Breakdown

1. Dijkstra Foundation: Standard Dijkstra to find the minimum distance from source to all nodes.
2. Path Counting: If we find a strictly shorter path to a node, we reset its path count to that of its parent. If we find a path of equal distance, we add the parent's path count to it.

Robust Version (Long Long)

class Solution {
public:
    int countPaths(int n, vector<vector<int>>& roads) {
        vector<vector<pair<int,int>>> adjList(n);
        const int MOD = 1e9 + 7;
        for(auto it: roads) {
            adjList[it[0]].push_back({it[1],it[2]});
            adjList[it[1]].push_back({it[0],it[2]});
        }

        vector<long long> dist(n,1e18);
        vector<long long> ways(n,0);
        int destination=n-1;
        priority_queue<pair<long long,int>, vector<pair<long long,int>>, greater<pair<long long,int>>> pq;

        pq.push({0,0});
        dist[0]=0;
        ways[0]=1;
        while(!pq.empty()) {
            auto[dist_,node]=pq.top();
            pq.pop();

            if (dist_ > dist[node]) continue;

            for(auto it: adjList[node]) {
                int adjNode = it.first;
                int edgeWeight = it.second;
                long long newdist=dist_+edgeWeight;
                if(newdist < dist[adjNode]) {
                    dist[adjNode]=newdist;
                    ways[adjNode]=ways[node];
                    pq.push({dist[adjNode],adjNode});
                }
                else if(newdist==dist[adjNode]) {
                    ways[adjNode] = (ways[adjNode]+ways[node])%MOD;                
                }
            }
        }
        return ways[destination];
    }
};

Simpler Version

class Solution {
public:
    int countPaths(int n, vector<vector<int>>& roads) {
        vector<vector<pair<int,int>>> adjList(n);
        for(auto it: roads) {
            adjList[it[0]].push_back({it[1], it[2]});
            adjList[it[1]].push_back({it[0], it[2]});
        }

        vector<int> dist(n, 1e9);  
        vector<int> ways(n, 0);    
        int destination = n - 1;

        priority_queue<pair<int,int>, vector<pair<int,int>>, greater<pair<int,int>>> pq;

        dist[0] = 0;
        ways[0] = 1;
        pq.push({0, 0});  

        while(!pq.empty()) {
            auto [dist_, node] = pq.top();
            pq.pop();

            if(dist_ > dist[node]) continue;

            for(auto it: adjList[node]) {
                int adjNode = it.first;
                int edgeWeight = it.second;

                if(dist_ + edgeWeight < dist[adjNode]) {
                    dist[adjNode] = dist_ + edgeWeight;
                    ways[adjNode] = ways[node];
                    pq.push({dist[adjNode], adjNode});
                } else if(dist_ + edgeWeight == dist[adjNode]) {
                    ways[adjNode] = ways[adjNode] + ways[node];
                }
            }
        }
        return ways[destination];
    }
};

Shortest Path in Binary Matrix

LeetCode 1091 | BFS on 2D Grid
Given an n x n binary grid, return the length of the shortest clear path from the top-left (0, 0) to the bottom-right (n-1, n-1). All 8 directions are allowed. A clear path is a path of 0s.
Time ComplexityO(N²)
Space ComplexityO(N²)

Logic Breakdown

1. 8-Way Traversal: Unlike standard grid problems, we use 8 directions (including diagonals) using the direction arrays.
2. BFS for Shortest Path: Since all edges have weight 1, BFS naturally finds the shortest distance first.
3. Edge Cases: Check if the source or destination is blocked (grid value is 1) at the start.
class Solution {
public:
    int shortestPathBinaryMatrix(vector<vector<int>>& grid) {
        int n=grid.size();
        vector<vector<int>> dist(n,vector<int> (n,1e9));

        if(grid[0][0]==1 || grid[n-1][n-1]==1)
        return -1;

        queue<pair<int,pair<int,int>>> q;
        // {dist,{row,col}}
        q.push({1,{0,0}});
        dist[0][0]=0;

        int dir_r[] = {-1,0,1,0,-1,0,-1,1};
        int dir_c[] = {0,1,0,-1,-1,-1,1,1};

        while(!q.empty())
        {
            auto it=q.front();
            int dist_=it.first;
            int row=it.second.first;
            int col=it.second.second;
            q.pop();

            // traverse the neighbors in all 8 directions
            for(int i=0;i<8;i++)
            {
                int newr = row+dir_r[i];
                int newc = col+dir_c[i];
                if(newr>=0 && newr<n && newc>=0 && newc<n
                && grid[newr][newc]==0 && dist_+1<dist[newr][newc])
                {
                    dist[newr][newc]=dist_+1;
                    q.push({dist_+1,{newr,newc}});
                }
            }
        }
        return dist[n-1][n-1]==1e9? -1: dist[n-1][n-1];
    }
};

Find the City With the Smallest Number of Neighbors

LeetCode 1334 | Floyd-Warshall vs. Multi-source Dijkstra
There are n cities numbered from 0 to n-1. Given the array edges where edges[i] = [from, to, weight], find the city with the smallest number of cities that are reachable through some path and whose distance is at most distanceThreshold. If there are multiple such cities, return the city with the greatest index.

The Floyd-Warshall Equation

This algorithm uses Dynamic Programming to find shortest paths between all pairs of vertices. The core recurrence relation is:

$$dist[i][j] = \min(dist[i][j], dist[i][k] + dist[k][j])$$

Alternatively, we can run Dijkstra $N$ times (once for each city) to achieve the same result with different performance characteristics.

Floyd-Warshall TimeO(N³)
Dijkstra TimeO(N * E log V)
Space ComplexityO(N²)

Logic Breakdown

1. Approach A (Floyd-Warshall): Maintain a 2D matrix. Iterate through all triplets $(k, i, j)$ to relax distances. Best for dense graphs.
2. Approach B (Dijkstra): Convert the edge list to an adjacency list. For every city (source), run a Priority Queue-based Dijkstra to find distances to all other nodes.
3. Filtering: In both cases, once the shortest path matrix is populated, count neighbors for each city where distance <= threshold.
4. Result: Return the largest city index among those with the minimum reachable neighbor count.

1. Floyd-Warshall Implementation

class Solution {
public:
    int findTheCity(int n, vector<vector<int>>& edges, int distanceThreshold) {
        vector<vector<int>> dist(n, vector<int> (n, INT_MAX));

        for(int i = 0; i < n; i++) dist[i][i] = 0;

        for(auto it : edges) {
            dist[it[0]][it[1]] = it[2];
            dist[it[1]][it[0]] = it[2];
        }

        for(int k = 0; k < n; k++) {
            for(int i = 0; i < n; i++) {
                for(int j = 0; j < n; j++) {
                    if(dist[i][k] == INT_MAX || dist[k][j] == INT_MAX)
                        continue;

                    dist[i][j] = min(dist[i][j], dist[i][k] + dist[k][j]);
                }
            }
        }

        int cntCity = INT_MAX;
        int cityNum = 0;

        for(int city = 0; city < n; city++) {
            int cnt = 0;
            for(int adjCity = 0; adjCity < n; adjCity++) {
                if(city != adjCity && dist[city][adjCity] <= distanceThreshold) {
                    cnt++;
                }
            }
            if(cnt <= cntCity) {
                cntCity = cnt;
                cityNum = city;
            }        
        }
        return cityNum;
    }
};

2. Multi-source Dijkstra Implementation

Dijkstra TimeO(V * (E log V))
Dijkstra SpaceO(V + E)
class Solution {
public:
    vector<int> dijkstra(int src,int n, vector<vector<pair<int,int>>> adjList)
    {
        vector<int> dist(n,INT_MAX);
        priority_queue<pair<int,int>, vector<pair<int,int>>, 
        greater<pair<int,int>>> pq;

        // {dist,node}
        pq.push({0,src});
        dist[src]=0;
        while(!pq.empty())
        {
            auto it=pq.top();
            int node=it.second;
            int dist_=it.first;
            pq.pop();
                        
            for(auto it: adjList[node])
            {
                int adjNode=it.first;
                int edgeWeight=it.second;

                if(dist_+edgeWeight < dist[adjNode])
                {
                    dist[adjNode]= dist_+edgeWeight;
                    pq.push({dist[adjNode],adjNode});
                }
            }
        }
        return dist;
    }

Min Cost to Connect Points

LeetCode 1584 | Prim's Algorithm
You are given n points on a 2D plane. You need to connect all points such that there is a path between any two points and the total cost (Manhattan Distance) is minimized.

Manhattan Distance: |x1 - x2| + |y1 - y2|
What is a Spanning Tree?Connects all nodes with NO cycles
What is an MST?The Spanning Tree with the LOWEST total weight
The "Greedy" ChoiceAlways pick the shortest available edge next

Prim's Intuition: The "Growing Ink Blot"

Imagine dropping ink on one point. The ink wants to spread to all other points using the shortest possible bridges.

  1. Start: Pick point 0 and put it in your "Ink Blot" (MST).
  2. Look: Check all bridges (edges) leading from the ink blot to "dry" points.
  3. Pick: Use a Min-Priority Queue to instantly find the absolute shortest bridge.
  4. Expand: Cross that bridge, add the new point to the ink blot, and repeat until all points are wet.

C++ Implementation

class Solution {
public:
    int minCostConnectPoints(vector<vector<int>>& points) {
        int n = points.size();
        // adjList[i] stores {neighbor_node, weight}
        vector<vector<pair<int,int>>> adjList(n);

        // 1. PRE-PROCESS: Calculate distance between every pair of points
        // In this problem, every point can connect to every other point (Dense Graph).
        for(int i = 0; i < n; i++) {
            for(int j = i + 1; j < n; j++) {
                int cost = abs(points[i][0] - points[j][0]) + 
                           abs(points[i][1] - points[j][1]);
                adjList[i].push_back({j, cost});
                adjList[j].push_back({i, cost});
            }
        }

        // 2. SETUP PRIM'S:
        // Priority Queue (PQ) acts as our 'Shortest Bridge Finder'.
        // It stores {weight, node} and keeps the smallest weight at the top.
        priority_queue<pair<int,int>, vector<pair<int,int>>, greater<pair<int,int>>> pq;

        vector<bool> inMST(n, false); // Tracks nodes already in our 'Ink Blot'
        int totalCost = 0;

        // Start with node 0 (weight 0 because we are already there)
        pq.push({0, 0});

        while(!pq.empty()) {
            auto it = pq.top();
            int cost = it.first;
            int node = it.second;
            pq.pop();

            // If node is already in MST, skip to avoid cycles
            if(inMST[node]) continue;

            // ADD TO MST: This node is now part of our minimum cost network
            inMST[node] = true;
            totalCost += cost;

            // Look at all potential bridges from this new node to its neighbors
            for(auto it : adjList[node]) {
                int adjNode = it.first;
                int edgeWeight = it.second;

                // If neighbor isn't in MST, add bridge to the Priority Queue
                if(!inMST[adjNode]) {
                    pq.push({edgeWeight, adjNode});
                }
            }
        }
        return totalCost;
    }
};

Required to store all N² possible edge costs

Min Cost to Connect Points

LeetCode 1584 | Kruskal's Algorithm
You are given n points on a 2D plane. You need to connect all points such that there is a path between any two points and the total cost (Manhattan Distance) is minimized.

Manhattan Distance: |x1 - x2| + |y1 - y2|
What is Kruskal's?A Greedy edge-based MST algorithm
Edge StrategySort ALL edges and pick the cheapest first
The "Cycle" GuardUse DSU to check if nodes are already connected

Kruskal's Intuition: The "Cheapest Bridge"

Imagine every point is a lonely island. You are a contractor trying to connect them all with the smallest possible budget.

  1. The Shopping List: First, calculate the cost to build a bridge between every single pair of points.
  2. Sorting: Sort this list of bridges from cheapest to most expensive. This is the Greedy heart of Kruskal's.
  3. The Connectivity Test: Pick the cheapest bridge. Before building it, check: "Are these two points already connected through other bridges we built?"
  4. Decision:
    • If No: Build the bridge! It's a bargain and adds a new connection.
    • If Yes: Skip it! Building it would create a Cycle (a useless loop that wastes money).
  5. Completion: Stop when you have successfully built $n - 1$ bridges.

DSU: The "Friendship Groups" Logic

Disjoint Set Union (DSU) is the tool that Kruskal uses to instantly check connectivity. It manages "groups" of connected points.

  • The Ultimate Parent (Boss): Every point belongs to a group. Each group has one "Big Boss" (Root). If two points have the same Big Boss, they are already connected.
  • Find Function: This asks a node, "Who is your Big Boss?"
    Path Compression: After finding the Boss, we connect the node directly to them. This turns a long chain of command into a flat team, making future lookups nearly instant!
  • Union Function: When we build a bridge, we merge two groups.
    Union by Rank: We always make the Boss of the smaller group report to the Boss of the larger group. This keeps our "organizational chart" short and efficient.

C++ Implementation

// 1. DSU CLASS: The "Group Manager"
class DSU {
public:
    vector<int> parent, rank;
    DSU(int n) {
        parent.resize(n); rank.resize(n, 0);
        for(int i = 0; i < n; i++) parent[i] = i; 
    }

    // Path Compression: Flatten the tree so every node points to the 'Big Boss'
    int findUltimateParent(int node) {
        if(parent[node] == node) return node;
        return parent[node] = findUltimateParent(parent[node]);
    }

    // Union by Rank: Keep the tree height small for speed
    void unionSet(int u, int v) {
        u = findUltimateParent(u); v = findUltimateParent(v);
        if(u != v) {
            if(rank[u] < rank[v]) swap(u, v);
            parent[v] = u;
            if(rank[u] == rank[v]) rank[u]++;
        }
    }
};

class Solution {
public:
    int minCostConnectPoints(vector<vector<int>>& points) {
        int n = points.size();
        vector<vector<int>> edges;

        // 2. PRE-PROCESS: Calculate distance between every pair of points
        for(int i = 0; i < n; i++) {
            for(int j = i + 1; j < n; j++) {
                int cost = abs(points[i][0] - points[j][0]) + 
                           abs(points[i][1] - points[j][1]);
                edges.push_back({cost, i, j});
            }
        }

        // 3. SORT: Cheapest bridges first (Kruskal's core strategy)
        sort(edges.begin(), edges.end());

        DSU dsu(n);
        int totalCost = 0, edgesUsed = 0;

        // 4. PROCESS: Pick edges until everything is connected
        for(auto &e : edges) {
            int cost = e[0], u = e[1], v = e[2];

            // Connect only if they are not already in the same group
            if(dsu.findUltimateParent(u) != dsu.findUltimateParent(v)) {
                dsu.unionSet(u, v);
                totalCost += cost;
                if(++edgesUsed == n - 1) break; 
            }
        }
        return totalCost;
    }
};

Number of Operations to Make Network Connected

LeetCode 1319 | Disjoint Set Union (DSU)
You are given n computers and a list of connections. You need to connect all computers into a single network. You can extract an existing cable between two directly connected computers and use it to connect any two disconnected computers.

Goal: Find the minimum number of extra cables needed to make all computers connected. If it's impossible, return -1.
Core StrategyIdentify Isolated Groups & Excess Wires
The RequirementN-1 edges are needed for N nodes
The FormulaOperations = (Total Components - 1)

Intuition: The "Redundant Cable" Strategy

To connect C separate clusters into one, you need exactly C - 1 additional cables. We find clusters and redundant wires simultaneously using DSU.

  1. Redundant Wires: Any edge that connects two nodes already in the same cluster is "extra."
  2. Clusters: The number of nodes that are their own parents after all unions represents the number of isolated networks.

C++ Implementation

class Solution {
public:
class DisjointSet{
    public: 
    int n;
    vector<int> parent;
    vector<int> rank;

    DisjointSet(int n_)
    {
        n=n_;
        parent.resize(n);
        rank.resize(n,0);
        for(int i=0;i<n;i++)
            parent[i]=i; // Initially, every node is its own parent (own group)
    }

    // Path Compression: Flattens the tree for near O(1) lookups
    int findUltimateParent(int node)
    {
        if(node==parent[node])
            return node;

        // Recursively find root and attach current node directly to it
        return parent[node]=findUltimateParent(parent[node]);
    }

    // Union by Rank: Attaches the smaller tree to the larger tree
    void unionBysize(int u,int v)
    {
        int pu = findUltimateParent(u);
        int pv = findUltimateParent(v);
        if(pu==pv) return; // Already in the same group

        if(rank[pu]>rank[pv])
        {
            parent[pv]=pu;
        }
        else if(rank[pv]>rank[pu])
        {
            parent[pu]=pv;
        }
        else
        {
            parent[pv]=pu;
            rank[pu]++; // Increase rank if heights are equal
        }
    }
};

int makeConnected(int n, vector<vector<int>>& connections) {
    DisjointSet ds(n);
    int cntExtras=0; // To store number of redundant cables

    // Step 1: Iterate through each connection
    for(auto it: connections)
    {
        int u = it[0];
        int v = it[1];

        // If they share the same root, this cable is redundant
        if(ds.findUltimateParent(u)==ds.findUltimateParent(v))
        {
            cntExtras++;
        }
        else
        {
            // Otherwise, merge the two groups
            ds.unionBysize(u,v);
        }
    }

    // Step 2: Count the total number of connected components
    int cntC=0;
    for(int i=0;i<n;i++)
    {
        // A node is a root if it points to itself
        if(ds.parent[i]==i)
            cntC++;
    }

    // Step 3: Check if we have enough extras to connect all components
    // Formula: Edges needed = Components - 1
    int ans=cntC-1;
    
    if(cntExtras>=ans)
        return ans;

    return -1; // Not enough redundant cables
}
};

Processing E edges and N nodes. DSU amortized time is constant.
Space ComplexityO(N)
DSU requires O(N) space for parent and rank arrays.

Strongly Connected Components (Kosaraju's Algorithm)

Graph Theory | Depth First Search (DFS) | Topological Sort
A Strongly Connected Component (SCC) is a portion of a directed graph where every node is reachable from every other node in that same portion.

Goal: Given a directed graph with V vertices and E edges, find the total number of Strongly Connected Components.
Core LogicThree-Step Graph Reversal
Primary ToolDFS & Stack (Finish Time)
Key BenefitLinear time discovery of clusters

Intuition: The "Point of No Return"

In a directed graph, if you can go from A to B and B back to A, they belong to the same SCC. Kosaraju’s uses a clever 3-step process to "trap" DFS inside these components:

  1. The Sorting (Finish Times): We perform a DFS. When a node's neighbors are fully explored, we push it onto a Stack. The nodes at the top of the stack are the ones that "finished" last (the start of a component chain).
  2. The Reverse (Transpose): We flip every edge in the graph $(u \to v$ becomes $v \to u)$. This ensures that DFS cannot "leak" from one SCC into another SCC that it previously could reach.
  3. The Isolation: We pop nodes from our stack. If a node isn't visited, it must be the start of a new SCC. We perform DFS on the reversed graph to visit everything reachable—which, thanks to the reversal, is only the current SCC.

Step-by-Step Pseudocode

  • Step 1: Perform DFS on the original graph. Once a node is finished, push it onto a Stack. This mimics topological sorting.
  • Step 2: Create a new adjacency list where all edge directions are reversed.
  • Step 3: While the stack is not empty:
    • Pop the top node.
    • If it hasn't been visited in the second pass, start a new DFS from it using the reversed graph.
    • Each successful DFS start counts as one SCC.

C++ Implementation

// Standard DFS to store nodes in stack according to finish time
void dfs(int node, vector<vector<int>> &adj, vector<int> &visited, stack<int> &st) {
    visited[node] = 1;
    for (auto it : adj[node]) {
        if (!visited[it]) {
            dfs(it, adj, visited, st);
        }
    }
    // Push to stack after visiting all children (Post-order)
    st.push(node);
}

// Second DFS to traverse the reversed (transposed) graph
void dfs2(int node, vector<vector<int>> &adj, vector<int> &visited) {
    visited[node] = 1;
    for (auto it : adj[node]) {
        if (!visited[it]) {
            dfs2(it, adj, visited);
        }
    }
}

int kosaraju(int V, vector<vector<int>> &adj) {
    stack<int> st;
    vector<int> visited(V, 0);

    // 1. Fill the stack based on finish times
    for (int i = 0; i < V; i++) {
        if (!visited[i]) {
            dfs(i, adj, visited, st);
        }
    }

    // 2. Transpose the graph (Reverse all edge directions)
    vector<vector<int>> transposedAdj(V);
    for (int i = 0; i < V; i++) {
        for (auto it : adj[i]) {
            transposedAdj[it].push_back(i);
        }
    }
    
    // 3. Perform DFS on the transposed graph in Stack order
    vector<int> visited2(V, 0);
    int sccCount = 0;
    
    while (!st.empty()) {
        int node = st.top();
        st.pop();
        
        if (!visited2[node]) {
            // This marks the discovery of a new SCC
            dfs2(node, transposedAdj, visited2);
            sccCount++;
        }
    }
    return sccCount;
}

Two DFS passes and one graph transposition.
Space ComplexityO(V + E)
Requires space for the stack, visited arrays, and transposed graph.

Theoretical Foundations, Complexity Classes, and Approximation Algorithms

1. Class P and 2. Class NP

Class P: Includes decision problems solvable by a deterministic algorithm in polynomial time $O(n^k)$. These are considered tractable or "easy" to solve.

Class NP: Includes problems where a given solution (certificate) can be verified in polynomial time. "NP" stands for Nondeterministic Polynomial time.

Example: GCD (Class P)

Problem: Find GCD(24, 18).

Algorithm: Euclidean Algorithm.

Dry Run:
Step 1: 24 divided by 18, remainder = 6.
Step 2: 18 divided by 6, remainder = 0.
Step 3: Remainder is 0, so GCD is 6.
Complexity: $O(\log(\min(a,b)))$

Example: Sudoku (Class NP)

Problem: Solve a 9x9 grid.

Verification Logic: If someone gives you a completed grid, you check each row, column, and 3x3 box for numbers 1-9. This takes roughly 81 checks.

Verification Complexity: $O(1)$ or $O(n^2)$ relative to board size.

3. NP-Hard and 4. NP-Complete (NPC)

NP-Hard: A problem is NP-Hard if every problem in NP can be reduced to it in polynomial time. These problems are "at least as hard" as the hardest problems in NP.

NP-Complete: A problem that is both in NP and is NP-Hard. It is the intersection of the two classes.

2. For each pair of vertices (u, v) in S:

5. return ACCEPT

5. Traveling Salesman Problem (TSP)

Definition: Given a set of cities and distances between them, find the shortest possible route that visits every city exactly once and returns to the starting city.

6. Reduction: Hamiltonian Cycle to TSP

Logic: We convert a Hamiltonian Cycle problem (Decision) into a TSP (Optimization) problem to prove TSP's complexity.

Dry Run of Reduction:
Suppose Graph G has 3 nodes {A, B, C} and edges (A,B), (B,C). (No edge A-C).
1. Create Complete Graph G' with all nodes connected.
2. Edge exists in G? Set weight = 1. (Weight A-B=1, B-C=1).
3. Edge missing in G? Set weight = 2. (Weight A-C=2).
4. TSP finds min tour in G': A-B-C-A = 1+1+2 = 4.
5. Since cost (4) > number of nodes (3), No Hamiltonian Cycle exists.

2. Set Cover Problem (Vertex Cover in Universal Set)

Problem: Given a universal set X and subsets F, find the smallest collection of subsets that covers X.

Example:
X = {a,b,c,d,e}
F = { f1:{a,b}, f2:{b,c}, f3:{c,d}, f4:{d,e}, f5:{a,e} }
Minimum Cover = {f1, f3, f4} → covers all elements.
Algorithm (Greedy Set Cover):
1. Pick the subset that covers the most uncovered elements.
2. Add to cover set.
3. Remove covered elements from X.
4. Repeat until X is empty.

Steps:

  1. Find a Minimum Spanning Tree (MST) of the graph.
  2. Choose any vertex as root.
  3. Perform a Preorder Tree Walk of MST to generate TSP tour.
  4. Total weight ≤ 2 × Optimal TSP (triangle inequality holds).

MST edges: A-B, B-C, B-D

7. Vertex Cover and 8. 2-Approximation

Vertex Cover: A set of vertices $V'$ such that every edge in the graph is incident to at least one vertex in $V'$.

Algorithm: 2-Approximation for Vertex Cover

Algorithm APPROX-VC(Graph G):
1. C = empty set
2. E' = copy of edges in G
3. While E' is not empty:
4.    Pick an arbitrary edge (u, v) from E'
5.    Add u and v to C
6.    Remove all edges from E' that are incident to u or v
7. Return C

9. Why 2-Approximation? This algorithm is used because Minimum Vertex Cover is NP-Hard. We pick both endpoints of an edge to guarantee we cover it. Since the optimal solution must pick at least one of those endpoints, our result is never more than twice the optimal size.

Dry Run (Vertex Cover):
Graph Edges: (1,2), (2,3), (3,4).

2. Remove all edges touching 1 or 2 (Removes (1,2) and (2,3)).
3. Remaining edges: (3,4).
4. Pick edge (3,4). Add {3, 4} to Set C.
5. Final Result: {1, 2, 3, 4}. (Note: Optimal was {2, 3} or {2, 4}).

Beginner Lecture: From Recursion to Optimization

1. What is Dynamic Programming?

DP is a technique to solve a big problem by breaking it into smaller subproblems, solving each subproblem only once, and reusing their answers.

Real-life Analogy: Imagine solving math homework. You solve Q5. Later, Q10 uses the result of Q5. Instead of re-calculating, you just "remember" the answer. That is DP.

2. Why Do We Need DP?

Because standard recursion repeats work unnecessarily. Look at the Fibonacci recursion tree for fib(5):

fib(5)
├── fib(4)
│   ├── fib(3)
│   │   ├── fib(2)
│   │   └── fib(1)
│   └── fib(2)
└── fib(3) (Computed Again!)

This leads to Exponential Time Complexity O(2ⁿ).

Core Requirement 1 Overlapping Subproblems Solving same thing again and again
Core Requirement 2 Optimal Substructure Big solution uses small solutions

4. The Fibonacci Problem: 4 Approaches

A. Recursion (The Bad Way) - Complexity: O(2ⁿ)
int fib(int n) {
    if (n <= 1) return n;
    return fib(n-1) + fib(n-2);
}
B. Memoization (Top-Down) - Complexity: O(N)
int fibMemo(int n, vector<int>& dp) {
    if (n <= 1) return n;
    if (dp[n] != -1) return dp[n]; // Reuse step

    return dp[n] = fibMemo(n-1, dp) + fibMemo(n-2, dp);
}
C. Tabulation (Bottom-Up) - Complexity: O(N)
int fib(int n) {
    vector<int> dp(n+1);
    dp[0] = 0; dp[1] = 1;
    for(int i=2; i<=n; i++) {
        dp[i] = dp[i-1] + dp[i-2];
    }
    return dp[n];
}
D. Space Optimization (The Best Way) - Complexity: Space O(1)
int fib(int n) {
    int prev2 = 0, prev1 = 1;
    for(int i=2; i<=n; i++) {
        int curr = prev1 + prev2;
        prev2 = prev1;
        prev1 = curr;
    }
    return prev1;
}

5. Exam Q&A

Q: Difference between Memoization and Tabulation?

Memoization is top-down (starts at N) and uses recursion. Tabulation is bottom-up (starts at 0) and uses loops. Tabulation usually has better space constant as it avoids the recursion stack.

Q: When is space optimization possible?

When current state dp[i] only depends on a constant number of previous states (like i-1 and i-2).

Climbing Stairs

LeetCode 70 | Fundamental Dynamic Programming
You are climbing a staircase. It takes n steps to reach the top. Each time you can either climb 1 or 2 steps. In how many distinct ways can you climb to the top?
Time ComplexityO(N)
Space ComplexityO(1) Optimized

How to Think: The DP Mental Framework

If you are a beginner, follow these 3 steps to "discover" the solution:

  1. The Goal: I want to reach step n.
  2. The "One Step Back" Rule: If I am standing on the top (step n), where could I have possibly come from?
    • I could have come from step n-1 (by taking a 1-step jump).
    • I could have come from step n-2 (by taking a 2-step jump).
  3. The Logic: Total ways to reach n = (Ways to reach n-1) + (Ways to reach n-2).
    Notice how this naturally reveals the Fibonacci pattern!

Approach 1: Tabulation (Bottom-Up)

class Solution {
public:
    int climbStairs(int n) {
        if(n <= 2) return n;
        vector<int> dp(n + 1);
        dp[1] = 1; // Base case: 1 step = 1 way
        dp[2] = 2; // Base case: 2 steps = 2 ways (1+1 or 2)
        for(int i = 3; i <= n; i++) {
            dp[i] = dp[i - 1] + dp[i - 2];
        }
        return dp[n];
    }
};

Approach 2: Memoization (Top-Down)

class Solution {
public:
    int climbStairs_(int n, vector<int> &dp) {
        if(n <= 2) return n;
        if(dp[n] != -1) return dp[n]; // Use previously calculated result
        
        return dp[n] = climbStairs_(n - 1, dp) + climbStairs_(n - 2, dp);
    }

    int climbStairs(int n) {
        vector<int> dp(n + 1, -1);
        return climbStairs_(n, dp);
    }
};

Approach 3: Space Optimization (Best)

Why optimize? Notice in Tabulation, to find dp[i], we only need the last two values (dp[i-1] and dp[i-2]). We don't need the whole array! 
class Solution {
public:
    int climbStairs(int n) {
        if(n <= 2) return n;
        int prev2 = 1; // Represents ways to reach step i-2
        int prev1 = 2; // Represents ways to reach step i-1
        for(int i = 3; i <= n; i++) {
            int curr = prev1 + prev2;
            prev2 = prev1; // Shift prev1 to prev2
            prev1 = curr;  // Shift current to prev1
        }
        return prev1;
    }
};

  • i=3: curr = 2 + 1 = 3 | Now prev2=2, prev1=3
  • i=4: curr = 3 + 2 = 5 | Now prev2=3, prev1=5
  • Result: 5

Frog Jump

GeeksforGeeks | Dynamic Programming on Arrays
You are given an array height[] of N stones. A frog is standing on stone 0 and wants to reach stone N-1. The frog can jump either 1 step or 2 steps at a time. The energy cost of a jump from stone i to stone j is |height[i] - height[j]|. Find the minimum total energy required to reach the last stone.
Time ComplexityO(N)
Space ComplexityO(N) (O(1) possible)

Deep Dive: How to Think about the Jumps

Imagine you are the frog standing on stone i. To minimize energy, you look backward to see how you could have arrived here:

  1. The 1-Step Jump: You were just at stone i-1 and jumped to i.
    Cost: Energy(i-1) + |height[i] - height[i-1]|
  2. The 2-Step Jump: You were at stone i-2 and skipped a stone to land on i.
    Cost: Energy(i-2) + |height[i] - height[i-2]|

Why do we use if (i > 1) for the Right Jump?

The variable i represents the current stone index. Dynamic Programming requires valid indices to access array values.

  • Index 0: The starting stone. Energy required is 0.
  • Index 1: The second stone. You can only come from Index 0 (a 1-step jump). A 2-step jump would require coming from Index -1, which does not exist.
  • Index 2 and beyond: Now, i > 1 is true. You can finally perform a 2-step jump because stone i-2 (Stone 0) actually exists.

Without this check, the code would try to access height[-1], causing a Runtime Error (Segmentation Fault).

Approach 1: Simple Recursion (Brute Force)

class Solution {
public:
    int f(int i, vector<int>& height) {
        // Base case: If we are at the first stone, 0 energy is used
        if (i == 0) return 0; 

        // Option 1: Jump from the immediate previous stone
        int left = f(i - 1, height) + abs(height[i] - height[i - 1]);
        
        int right = INT_MAX;
        // Option 2: Jump from two stones back (Check validity)
        if (i > 1) {
            right = f(i - 2, height) + abs(height[i] - height[i - 2]);
        }
        
        return min(left, right);
    }

    int minimumEnergy(vector<int>& height, int n) {
        return f(n - 1, height);
    }
};

Approach 2: Memoization (Top-Down)

class Solution {
public:
    int f(int i, vector<int>& height, vector<int>& dp) {
        if (i == 0) return 0;
        // Step 1: Check if result is already in memory
        if (dp[i] != -1) return dp[i];

        int left = f(i - 1, height, dp) + abs(height[i] - height[i - 1]);
        int right = INT_MAX;
        if (i > 1) {
            right = f(i - 2, height, dp) + abs(height[i] - height[i - 2]);
        }
        // Step 2: Store and return result
        return dp[i] = min(left, right);
    }

    int minimumEnergy(vector<int>& height, int n) {
        vector<int> dp(n, -1);
        return f(n - 1, height, dp);
    }
};

Approach 3: Tabulation (Bottom-Up)

class Solution {
public:
    int minimumEnergy(vector<int>& height, int n) {
        vector<int> dp(n);
        dp[0] = 0; // Base case

        for (int i = 1; i < n; i++) {
            int jumpOne = dp[i - 1] + abs(height[i] - height[i - 1]);
            int jumpTwo = INT_MAX;
            
            // Again, check i > 1 to avoid accessing dp[-1]
            if (i > 1) {
                jumpTwo = dp[i - 2] + abs(height[i] - height[i - 2]);
            }
            dp[i] = min(jumpOne, jumpTwo);
        }
        return dp[n - 1];
    }
};

Input: height = [10, 20, 30, 10]

  • i = 0: dp[0] = 0 (Starting point)
  • i = 1:
    1-step from Stone 0: 0 + |20-10| = 10
    2-step: INVALID (i not > 1)
    dp[1] = 10
  • i = 2:
    1-step from Stone 1: dp[1] + |30-20| = 10+10 = 20
    2-step from Stone 0: dp[0] + |30-10| = 0+20 = 20
    dp[2] = min(20, 20) = 20
  • i = 3:
    1-step from Stone 2: dp[2] + |10-30| = 20+20 = 40
    2-step from Stone 1: dp[1] + |10-20| = 10+10 = 20
    dp[3] = min(40, 20) = 20

Final Answer: 20

Frog Jump with K Jumps

Dynamic Programming | Generalization of Frog Jump
You are given an array height[] of N stones and an integer K. A frog starts at stone 0 and wants to reach stone N-1. The frog can jump any number of steps from 1 to K at a time. The energy cost of a jump from stone i to stone j is |height[i] - height[j]|. Find the minimum total energy required to reach the last stone.
Time ComplexityO(N * K)
Space ComplexityO(N)

Intuition: From 2 Choices to K Choices

In the basic Frog Jump, the frog could only look back at 2 stones. Now, the frog looks back at K stones to find the cheapest way to arrive at the current stone i.

Recurrence Relation:

dp[i] = min(dp[i - j] + abs(height[i] - height[i - j]))
for all j where 1 ≤ j ≤ K and i - j ≥ 0

The "For Loop" Strategy

Instead of hardcoding left and right jumps, we use a loop that runs K times for every stone. This makes the algorithm flexible for any jump limit.

Approach 1: Memoization (Top-Down)

class Solution {
public:
    int f(int i, vector<int>& height, int k, vector<int>& dp) {
        if (i == 0) return 0;
        if (dp[i] != -1) return dp[i];

        int minSteps = INT_MAX;

        // Try every possible jump from 1 to K
        for (int j = 1; j <= k; j++) {
            if (i - j >= 0) {
                int jump = f(i - j, height, k, dp) + abs(height[i] - height[i - j]);
                minSteps = min(minSteps, jump);
            }
        }
        return dp[i] = minSteps;
    }

    int minimizeCost(vector<int>& height, int n, int k) {
        vector<int> dp(n, -1);
        return f(n - 1, height, k, dp);
    }
};

Approach 2: Tabulation (Bottom-Up)

class Solution {
public:
    int minimizeCost(vector<int>& height, int n, int k) {
        vector<int> dp(n, INT_MAX);
        dp[0] = 0;

        for (int i = 1; i < n; i++) {
            // For each stone i, look back at stones from i-1 to i-k
            for (int j = 1; j <= k; j++) {
                if (i - j >= 0) {
                    int cost = dp[i - j] + abs(height[i] - height[i - j]);
                    dp[i] = min(dp[i], cost);
                }
            }
        }
        return dp[n - 1];
    }
};

Dry Run Trace:

Input: height = [10, 30, 40, 20], K = 2

  • i = 0: dp[0] = 0
  • i = 1:
    Jump 1: dp[0] + |30-10| = 20
    dp[1] = 20
  • i = 2:
    Jump 1 (from i=1): dp[1] + |40-30| = 20+10 = 30
    Jump 2 (from i=0): dp[0] + |40-10| = 0+30 = 30
    dp[2] = min(30, 30) = 30
  • i = 3:
    Jump 1 (from i=2): dp[2] + |20-40| = 30+20 = 50
    Jump 2 (from i=1): dp[1] + |20-30| = 20+10 = 30
    dp[3] = min(50, 30) = 30

Final Answer: 30

Maximum Sum of Non-Adjacent Elements

LeetCode 198 | House Robber | Classical DP
You are given an array of integers arr[]. Find the maximum possible sum such that no two chosen elements are adjacent.

Analogy: A robber wants to rob houses along a street. Each house has a certain amount of money. The only constraint is that he cannot rob two adjacent houses.
Time ComplexityO(N)
Space ComplexityO(1) Optimized

DP Intuition: Pick or Not Pick?

At every index i, you have exactly two choices that define the optimal substructure:

  1. Pick current element: You gain arr[i], but you must skip the adjacent house. So, you look at the best sum up to i-2.
    Formula: arr[i] + dp[i-2]
  2. Do NOT pick current: You skip arr[i]. The best sum remains whatever the best was up to the previous house.
    Formula: dp[i-1]

Recurrence Relation: dp[i] = max(arr[i] + dp[i-2], dp[i-1])

Approach 1: Recursion + Memoization (Top-Down)

class Solution {
public:
    int f(int i, vector<int>& arr, vector<int>& dp) {
        if (i == 0) return arr[0]; // Base: only one house
        if (i < 0) return 0;      // Base: out of bounds

        if (dp[i] != -1) return dp[i];

        int pick = arr[i] + f(i - 2, arr, dp);
        int notPick = f(i - 1, arr, dp);

        return dp[i] = max(pick, notPick);
    }

    int rob(vector<int>& arr) {
        int n = arr.size();
        vector<int> dp(n, -1);
        return f(n - 1, arr, dp);
    }
};

Approach 2: Tabulation (Bottom-Up)

class Solution {
public:
    int rob(vector<int>& arr) {
        int n = arr.size();
        if (n == 1) return arr[0];

        vector<int> dp(n);
        dp[0] = arr[0];
        dp[1] = max(arr[0], arr[1]);

        for (int i = 2; i < n; i++) {
            int pick = arr[i] + dp[i - 2];
            int notPick = dp[i - 1];
            dp[i] = max(pick, notPick);
        }
        return dp[n - 1];
    }
};

Approach 3: Space Optimized (The Professional Way)

Logic: Since we only look back at dp[i-1] and dp[i-2], we can replace the entire array with just two variables: prev1 and prev2. 
class Solution {
public:
    int rob(vector<int>& arr) {
        int n = arr.size();
        int prev2 = 0;        // Think of this as dp[i-2]
        int prev1 = arr[0];   // Think of this as dp[i-1]

        for (int i = 1; i < n; i++) {
            int pick = arr[i] + prev2;
            int notPick = prev1;

            int curr = max(pick, notPick);
            prev2 = prev1;
            prev1 = curr;
        }
        return prev1;
    }
};

0 2 - - 2
1 1 1 + 0 = 1 2 2
2 4 4 + 2 = 6 2 6
3 9 9 + 2 = 11 6 11

Final Answer: 11 (Path: Pick 2 and 9)

House Robber II

LeetCode 213 | DP on Circular Arrays
You are a professional robber planning to rob houses along a street. Each house has a certain amount of money stashed. However, all houses are arranged in a circle. The first house is the neighbor of the last house. You cannot rob two adjacent houses. Calculate the maximum money you can rob without alerting the police.
Time ComplexityO(N)
Space ComplexityO(N) (O(1) possible)

Intuition: Breaking the Circle

Because the first and last houses are neighbors, you have a new constraint: You cannot rob both House 0 and House N-1.

To solve this, we convert the circular problem into two linear problems:

  1. Case 1: Rob houses while excluding the last house (indices 0 to n-2).
  2. Case 2: Rob houses while excluding the first house (indices 1 to n-1).

Final Answer: max(Result_of_Case1, Result_of_Case2)

Approach: Tabulation with Subarray Splitting

class Solution {
public:
    // Standard Linear House Robber Logic
    int robdp(vector<int>& nums) {
        int n = nums.size();
        if(n == 0) return 0;
        if(n == 1) return nums[0];
        
        vector<int> dp(n, 0);
        dp[0] = nums[0];
        dp[1] = max(nums[0], nums[1]);
        
        for(int i = 2; i < n; i++) {
            int take = nums[i] + dp[i - 2];   // Rob current
            int notTake = dp[i - 1];          // Skip current
            dp[i] = max(take, notTake);
        }
        return dp[n - 1];
    }

    int rob(vector<int>& nums) {
        int n = nums.size();
        if(n == 1) return nums[0];
        
        // Case 1: Exclude last house
        vector<int> temp1(nums.begin(), nums.end() - 1);
        // Case 2: Exclude first house
        vector<int> temp2(nums.begin() + 1, nums.end());
        
        return max(robdp(temp1), robdp(temp2));
    }
};

Dry Run Example

Input: nums = [2, 3, 2]

  • Step 1: Split into two arrays:
    • temp1 = [2, 3] (Exclude last)
    • temp2 = [3, 2] (Exclude first)
  • Step 2: Run linear rob logic:
    • robdp([2, 3]):
      dp[0]=2, dp[1]=max(2,3)=3. Result = 3.
    • robdp([3, 2]):
      dp[0]=3, dp[1]=max(3,2)=3. Result = 3.
  • Final result: max(3, 3) = 3

Note: We cannot rob the first '2' and the last '2' together because they are neighbors in the circle.

Ninja Training – Complete Guide

LeetCode / Coding Ninjas | Multi-Dimensional Dynamic Programming

1. Problem Statement

A ninja trains for n days. Each day, he can perform one of three activities: Activity 0, Activity 1, or Activity 2. Each activity gives different points, represented in a 2D array: points[day][activity].

Constraints:

  • The ninja cannot perform the same activity on two consecutive days.
  • Find the maximum total points he can earn in n days.
Example:
n = 3
points = [[10, 40, 70], [20, 50, 80], [30, 60, 90]]
Output: 210 (70 + 50 + 90)

2. Intuition

Each day, the choice depends on yesterday’s activity. We cannot simply pick the max points each day (greedy fails) because picking a high-value task today might force us to pick a very low-value task tomorrow.

We need Dynamic Programming to track the maximum points earned up to day i, given that a specific activity was performed on day i-1.

3. DP State & Recursive Approach

Define the DP state as: dp[day][last]

  • day: Current day (0 to n-1).
  • last: Activity done on the previous day (0, 1, 2). Use 3 for no restriction (first day).

Recursive Relation:
f(day, last) = max(points[day][task] + f(day-1, task)) for all task != last

Base Case (Day 0):
f(0, last) = max(points[0][task]) for all task != last

4. Memoization (Top-Down DP)

int f(int day, int last, vector<vector<int>> &points, vector<vector<int>> &dp) {
    // Base Case
    if(day == 0) {
        int maxi = 0;
        for(int task = 0; task < 3; task++)
            if(task != last) maxi = max(maxi, points[0][task]);
        return maxi;
    }

    if(dp[day][last] != -1) return dp[day][last];

    int maxi = 0;
    for(int task = 0; task < 3; task++) {
        if(task != last) {
            int point = points[day][task] + f(day-1, task, points, dp);
            maxi = max(maxi, point);
        }
    }
    return dp[day][last] = maxi;
}

int ninjaTraining(int n, vector<vector<int>> &points) {
    vector<vector<int>> dp(n, vector<int>(4, -1));
    return f(n-1, 3, points, dp);
}

5. Tabulation (Bottom-Up DP)

int ninjaTraining(int n, vector<vector<int>> &points) {
    vector<vector<int>> dp(n, vector<int>(4, 0));

    // Base case: day 0
    dp[0][0] = max(points[0][1], points[0][2]);
    dp[0][1] = max(points[0][0], points[0][2]);
    dp[0][2] = max(points[0][0], points[0][1]);
    dp[0][3] = max({points[0][0], points[0][1], points[0][2]});

    for(int day = 1; day < n; day++) {
        for(int last = 0; last < 4; last++) {
            for(int task = 0; task < 3; task++) {
                if(task != last) {
                    dp[day][last] = max(dp[day][last], points[day][task] + dp[day-1][task]);
                }
            }
        }
    }
    return dp[n-1][3];
}

6. Dry Run Example

Input: points = [[10,40,70],[20,50,80],[30,60,90]]

  • Day 0:
    dp[0][0] = max(40, 70) = 70 | dp[0][1] = 70 | dp[0][2] = 40.
  • Day 1:
    If we pick Task 0 (20 pts), we add dp[0][0] (70) = 90.
    If we pick Task 1 (50 pts), we add dp[0][1] (70) = 120.
    If we pick Task 2 (80 pts), we add dp[0][2] (40) = 120.
  • Day 2: Repeating logic leads to the maximum possible sum: 210.

7. Step-by-Step Explanation

  1. Recognize constraints: You cannot repeat the same activity on consecutive days.
  2. Define state: dp[day][last] stores max points until 'day' if the previous activity was 'last'.
  3. Recursive relation: Try all activities except the one done yesterday, add points from the current day + the optimal result from the previous day.
  4. Base case: At day 0, pick the maximum activity that isn't the restricted 'last' activity.
  5. Memoization: Use a 2D array to store results of (day, last) to prevent redundant calculations.
  6. Tabulation: Convert to an iterative approach building from day 0 up to n-1.
  7. Final Answer: Return dp[n-1][3] (where 3 means no restriction on the very last day).
Time ComplexityO(N * 4 * 3) ≈ O(N)
Space ComplexityO(N * 4)

Unique Paths

LeetCode 62 | 2D Dynamic Programming | Grid Traversal

1. Problem Statement

There is a robot on an m x n grid. The robot is initially located at the top-left corner (grid[0][0]). The robot tries to move to the bottom-right corner (grid[m-1][n-1]).

Constraints:

  • The robot can only move either down or right at any point in time.
  • Given the two integers m and n, return the number of possible unique paths.
Example:
m = 3, n = 2
Output: 3
Paths: 1. Right -> Down -> Down | 2. Down -> Down -> Right | 3. Down -> Right -> Down

2. Intuition

Since the robot can only move Down or Right, we can think about this problem backward. To arrive at any cell (i, j), the robot must have come from either:

  • The cell directly Above: (i-1, j)
  • The cell directly to the Left: (i, j-1)

Therefore, Paths(i, j) = Paths(i-1, j) + Paths(i, j-1). This overlapping nature makes it a perfect candidate for Dynamic Programming.

3. DP State & Recursive Approach

Define the DP state as: f(row, col)

  • row, col: The current coordinates of the robot in the grid.
  • Goal: Find f(m-1, n-1) starting from the destination and working back to the source.

Recursive Relation:
f(row, col) = f(row-1, col) + f(row, col-1)

Base Cases:
1. Source Reached: If row == 0 && col == 0, we found 1 valid path. Return 1.
2. Out of Bounds: If row < 0 || col < 0, the robot moved outside the grid. Return 0.

4. Memoization (Top-Down DP)

int f(int row, int col, vector<vector<int>> &dp) {
    // Base Case: Robot reached the top-left corner
    if(row == 0 && col == 0) 
        return 1;

    // Base Case: Robot went out of grid boundaries
    if(row < 0 || col < 0) 
        return 0;

    // Check if result is already computed (Memoization)
    if(dp[row][col] != -1) 
        return dp[row][col];

    // Recursive calls: Move Up and Move Left
    int up = f(row - 1, col, dp);
    int left = f(row, col - 1, dp);

    // Store the sum in the DP table and return
    return dp[row][col] = up + left;
}

int uniquePaths(int m, int n) {
    vector<vector<int>> dp(m, vector<int>(n, -1));
    return f(m - 1, n - 1, dp);
}

5. How Top-Down Memoization Works

Top-Down DP effectively prunes the recursion tree. Instead of recalculating identical sub-problems, the robot checks its "diary" (the DP table).

  • Step-by-Step Flow:
    1. Robot starts at (m-1, n-1) and explores backward.
    2. The 'Diary' Check: Before every call, it checks dp[row][col]. If it is not -1, it returns the result immediately in O(1).
    3. Saving Time: Without memoization, a 10x10 grid would take billions of calls. With it, we only solve 100 cells.

6. Tabulation (Bottom-Up DP)

In Tabulation, we remove recursion entirely. We build the solution from the base case (0,0) up to the target (m-1, n-1) using iterative loops.

int uniquePaths(int m, int n) {
    vector<vector<int>> dp(m, vector<int> (n, -1));
    
    // 1. Initialize the starting point (Base Case)
    dp[0][0] = 1;

    // 2. Iterate through the entire grid
    for(int i = 0; i < m; i++) {
        for(int j = 0; j < n; j++) {
            
            // Already handled (0,0)
            if(i == 0 && j == 0)
                dp[i][j] = 1;

            else {
                int up = 0;
                int left = 0;

                // Check if there is a cell above
                if(i > 0) up = dp[i - 1][j];

                // Check if there is a cell to the left
                if(j > 0) left = dp[i][j - 1];

                // Cell value = ways from above + ways from left
                dp[i][j] = up + left;
            }
        }
    }
    return dp[m - 1][n - 1];
}
Time ComplexityO(M × N)
Two nested loops visiting every cell once.
Space ComplexityO(M × N)
We only use the 2D DP table. No recursion stack.

Minimum Path Sum

Dynamic Programming on 2D Grids | Path Optimization
Given a m x n grid filled with non-negative numbers, find a path from top left to bottom right, which minimizes the sum of all numbers along its path.

Note: You can only move either down or right at any point in time.
Time ComplexityO(M * N)
Space ComplexityO(M * N)

Intuition: Minimum of Two Directions

To reach a cell (i, j), you can only arrive from the cell above (i-1, j) or the cell to the left (i, j-1).

To minimize the cost to reach (i, j), we should choose the minimum cost between these two possible origins and then add the current cell's value.

Recurrence Relation:

f(i, j) = grid[i][j] + min(f(i-1, j), f(i, j-1))

Handling Boundaries

If we are at (0, 0), the cost is simply grid[0][0]. If i < 0 or j < 0, we return a very large value (infinity) so that min() ignores that path.

Approach 1: Memoization (Top-Down)

class Solution {
public:
    int f(int i, int j, vector<vector<int>>& grid, vector<vector<int>>& dp) {
        if (i == 0 && j == 0)
            return grid[0][0];

        if (i < 0 || j < 0)
            return INT_MAX;

        if (dp[i][j] != -1)
            return dp[i][j];

        int up = f(i - 1, j, grid, dp);
        int left = f(i, j - 1, grid, dp);

        int ans = min(up, left);

        if (ans == INT_MAX)
            return dp[i][j] = INT_MAX;

        return dp[i][j] = grid[i][j] + ans;
    }

    int minPathSum(vector<vector<int>>& grid) {
        int m = grid.size();
        int n = grid[0].size();

        vector<vector<int>> dp(m, vector<int>(n, -1));

        return f(m - 1, n - 1, grid, dp);
    }
};

DP on Subsequences

The Master Guide to Subset and Subsequence Optimization

The Core Idea: Take vs. Not Take

In subsequence problems, for every element, you have a binary choice that determines the state transitions:

  • 1. Take: Include current element in sum/target, reduce target, and move to next index.
  • 2. Not Take: Ignore current element, keep target same, and move to next index.

1. Subset Sum (Existence)

Problem Statement: Given an array arr[] of non-negative integers and a target, determine if there exists any subsequence whose sum equals the target.

Intuition: At any index i, we have a target remaining. If we don't take arr[i], we still need to find target in the remaining array. If we do take it, we now need to find target - arr[i].
Recurrence: f(i, target) = f(i-1, target) || f(i-1, target - arr[i])
TimeO(N * Target)
SpaceO(N * Target)

Approach 1: Memoization

bool f(int i, int target, vector<int>& arr, vector<vector<int>>& dp) {
    if (target == 0) return true; // Found the sum
    if (i == 0) return arr[0] == target; // Last chance to match
    
    if (dp[i][target] != -1) return dp[i][target];

    bool notTake = f(i - 1, target, arr, dp);
    bool take = false;
    if (arr[i] <= target) 
        take = f(i - 1, target - arr[i], arr, dp);

    return dp[i][target] = take || notTake;
}

Detailed Dry Run (arr: [2, 3, 1], target: 4)

1. f(2, 4): Choices are f(1, 4) [Not Take 1] or f(1, 3) [Take 1].

2. f(1, 3): Choices are f(0, 3) [Not Take 3] or f(0, 0) [Take 3].

3. f(0, 0): Hits target == 0. Returns true.

4. Result propagates up: f(1, 3) is true, f(2, 4) is true.

2. Count Subsets with Sum K

Problem Statement: Find the total number of subsequences that sum up to exactly k.

Intuition: Same as existence, but instead of using "OR" (||), we use "ADDITION" (+) because we want to sum up all valid paths to the target.
Crucial Base Case: If arr[i] can be 0:
• If target=0 and arr[0]=0: 2 ways (Take 0 or Not Take 0).
• If target=0 or target=arr[0]: 1 way.

Approach 2: Tabulation

int findWays(vector<int>& arr, int k) {
    int n = arr.size();
    vector<vector<int>> dp(n, vector<int>(k + 1, 0));
    
    for (int i = 0; i < n; i++) dp[i][0] = 1; // Sum 0 is always 1 way
    if (arr[0] <= k) dp[0][arr[0]] = 1;

    for (int i = 1; i < n; i++) {
        for (int target = 1; target <= k; target++) {
            int notTake = dp[i - 1][target];
            int take = 0;
            if (arr[i] <= target) 
                take = dp[i - 1][target - arr[i]];
            
            dp[i][target] = take + notTake;
        }
    }
    return dp[n - 1][k];
}

0/1 Knapsack Patterns

Mastering the "Take or Not Take" Optimization Strategy

1. What is the 0/1 Knapsack Problem?

You are given N items, each with a specific weight and value. You have a bag (knapsack) with a limited capacity W.
The goal is to maximize the total value in the bag such that the sum of weights is ≤ W.

Why "0/1"? You cannot break items. For every item, you have exactly two choices:
0: Leave the item.
1: Take the item.
Time ComplexityO(N * W)
Space ComplexityO(N * W)
Optimized SpaceO(W)

2. How to Fill the DP Table

The DP table dp[i][w] represents the maximum value we can get using items from index 0 to i with a bag capacity of w.

The Core Recurrence Relation:

dp[i][w] = max( dp[i-1][w], val[i] + dp[i-1][w - wt[i]] )

  • Option 1 (Exclude): dp[i-1][w] → We don't take the current item. The value is whatever we could get with previous items at the same capacity.
  • Option 2 (Include): val[i] + dp[i-1][w - wt[i]] → We take current item. We gain its value, but our remaining capacity becomes w - wt[i]. We look at the best previous value for that remaining capacity.

3. Common Pattern Recurrences

Pattern Goal Recurrence Relation
Max Value Maximize Profit dp[i][w] = max(dp[i-1][w], val[i] + dp[i-1][w-wt[i]])
Subset Sum Existence (T/F) dp[i][target] = dp[i-1][target] || dp[i-1][target-arr[i]]
Count Subsets Total Ways dp[i][target] = dp[i-1][target] + dp[i-1][target-arr[i]]
Min Coins Optimization dp[i][w] = min(dp[i-1][w], 1 + dp[i][w-coin[i]])

4. Tabulation Approach (C++)

int knapsackTab(int n, int W, vector<int>& wt, vector<int>& val) {
    // 1. Initialize DP table with 0s
    vector<vector<int>> dp(n, vector<int>(W + 1, 0));

    // 2. Base Case: Fill first row for the first item
    for (int w = wt[0]; w <= W; w++) {
        dp[0][w] = val[0];
    }

    // 3. Fill the rest of the table row by row
    for (int i = 1; i < n; i++) {
        for (int cap = 0; cap <= W; cap++) {
            // Not Take Choice
            int notTake = dp[i - 1][cap];
            
            // Take Choice
            int take = INT_MIN;
            if (wt[i] <= cap) {
                take = val[i] + dp[i - 1][cap - wt[i]];
            }
            
            // Fill current cell with the best choice
            dp[i][cap] = max(take, notTake);
        }
    }
    return dp[n - 1][W];
}

Space Optimization Strategy

Since we only ever look at the previous row (i-1), we don't need a 2D matrix. We can use two rows or even a single row updated in reverse.

// One-row optimization (Space: O(W))
vector<int> prev(W + 1, 0);
for (int cap = W; cap >= wt[i]; cap--) {
    prev[cap] = max(prev[cap], val[i] + prev[cap - wt[i]]);
}

class Solution {

Shortest Common Supersequence

LeetCode 1092 | String Construction from LCS Table

Problem Description

Given two strings str1 and str2, return the shortest string that has both str1 and str2 as subsequences.

The Formula:
Length of SCS = (Length of Str1 + Length of Str2) - Length of LCS(Str1, Str2)

Intuition: Why use LCS?

To make the supersequence shortest, we want to write the common characters (the LCS) only once. Everything else that is not common must be included in the order it appears.

The Strategy: 1. Build the LCS DP Table. 2. Backtrack from dp[n][m] to dp[0][0] to construct the string.

Time ComplexityO(N × M)
Space ComplexityO(N × M)

1. Complete Implementation (Tabulation + Backtracking)

class Solution {
public:
    string shortestCommonSupersequence(string s1, string s2) {
        int n = s1.length();
        int m = s2.length();
        
        // Step 1: Standard LCS Tabulation
        vector<vector<int>> dp(n + 1, vector<int>(m + 1, 0));
        for (int i = 1; i <= n; i++) {
            for (int j = 1; j <= m; j++) {
                if (s1[i - 1] == s2[j - 1])
                    dp[i][j] = 1 + dp[i - 1][j - 1];
                else
                    dp[i][j] = max(dp[i - 1][j], dp[i][j - 1]);
            }
        }

        // Step 2: Backtrack to build the string
        string ans = "";
        int i = n, j = m;
        while (i > 0 && j > 0) {
            if (s1[i - 1] == s2[j - 1]) {
                ans += s1[i - 1]; // Common char, take once
                i--; j--;
            } else if (dp[i - 1][j] > dp[i][j - 1]) {
                ans += s1[i - 1]; // Take char from s1
                i--;
            } else {
                ans += s2[j - 1]; // Take char from s2
                j--;
            }
        }

        // Add remaining characters
        while (i > 0) { ans += s1[i - 1]; i--; }
        while (j > 0) { ans += s2[j - 1]; j--; }

        reverse(ans.begin(), ans.end());
        return ans;
    }
};

Step-by-Step Backtracking Logic

  1. Start at the bottom-right cell dp[n][m].
  2. If Characters Match (s1[i-1] == s2[j-1]):
    Move diagonally ↖. Add character to ans once.
  3. If Characters DON'T Match:
    Compare dp[i-1][j] (up) and dp[i][j-1] (left).
    Move to the cell with the larger value.
    Crucially: Add the character of the string you are "leaving behind" to ans.
  4. If one string is exhausted, append all remaining characters of the other string.
  5. Reverse the final ans string.

Problem Description

Given two strings str1 and str2, return the minimum number of operations required to transform str1 into str2. The allowed operations are:

  • Insertion: Add a character to the string.
  • Deletion: Remove a character from the string.
  • Substitution: Replace one character with another.
Goal: Find the minimum cost to make two strings identical.
Time ComplexityO(N × M)
Space ComplexityO(N × M)
ExampleKITTEN → SITTING

The Core Recurrence Logic

For each cell dp[i][j], we compare str1[i-1] and str2[j-1]:

  • If Match: dp[i][j] = dp[i-1][j-1] (Take the diagonal value).
  • If No Match: dp[i][j] = 1 + min(diagonal, above, left)
    • Diagonal (dp[i-1][j-1]): Substitution
    • Above (dp[i-1][j]): Deletion
    • Left (dp[i][j-1]): Insertion

1. Recursive Approach (Brute Force)

int solve(int i, int j, string& s1, string& s2) {
    // Base Case: If s1 is empty, insert all of s2
    if (i < 0) return j + 1;
    // Base Case: If s2 is empty, delete all of s1
    if (j < 0) return i + 1;

    if (s1[i] == s2[j]) 
        return solve(i - 1, j - 1, s1, s2);

    return 1 + min({
        solve(i - 1, j - 1, s1, s2), // Substitution
        solve(i - 1, j, s1, s2),     // Deletion
        solve(i, j - 1, s1, s2)      // Insertion
    });
}

2. Memoization Approach (Top-Down)

int solve(int i, int j, string& s1, string& s2, vector<vector<int>>& dp) {
    if (i < 0) return j + 1;
    if (j < 0) return i + 1;
    if (dp[i][j] != -1) return dp[i][j];

    if (s1[i] == s2[j]) 
        return dp[i][j] = solve(i - 1, j - 1, s1, s2, dp);

    return dp[i][j] = 1 + min({
        solve(i - 1, j - 1, s1, s2, dp), 
        solve(i - 1, j, s1, s2, dp), 
        solve(i, j - 1, s1, s2, dp)
    });
}

3. Tabulation Approach (Bottom-Up)

The tabulation approach fills the table iteratively from 0 up to n and m, avoiding the overhead of recursion. 
class Solution {
public:
    int minDistance(string s1, string s2) {
        int n = s1.length();
        int m = s2.length();
        // (n+1) x (m+1) table
        vector<vector<int>> dp(n + 1, vector<int>(m + 1, 0));

        // Base Cases
        for (int i = 0; i <= n; i++) dp[i][0] = i; // i deletions
        for (int j = 0; j <= m; j++) dp[0][j] = j; // j insertions

        for (int i = 1; i <= n; i++) {
            for (int j = 1; j <= m; j++) {
                if (s1[i - 1] == s2[j - 1])
                    dp[i][j] = dp[i - 1][j - 1];
                else
                    dp[i][j] = 1 + min({
                        dp[i - 1][j - 1], // Substitution
                        dp[i - 1][j],     // Deletion
                        dp[i][j - 1]      // Insertion
                    });
            }
        }
        return dp[n][m];
    }
};

int matrixChain(vector<int>& arr, int n) {
    // dp[n][n] stores the min costs
    vector<vector<int>> dp(n, vector<int>(n, 0));

    // len is the chain length
    for (int len = 2; len < n; len++) {
        for (int i = 1; i + len - 1 < n; i++) {
            int j = i + len - 1;
            dp[i][j] = INT_MAX;

            for (int k = i; k < j; k++) {
                int cost = dp[i][k] + dp[k+1][j]
                         + arr[i-1] * arr[k] * arr[j];
                dp[i][j] = min(dp[i][j], cost);
            }
        }
    }
    return dp[1][n-1];
}

Insertion Sort

Adjacent Swapping Implementation | Incremental Sorting

Problem Description

Sort an array of n elements by shifting each element to its correct position within the sorted prefix of the array. This implementation uses a j = i index and adjacent swaps to move the current element backward until it finds its correct spot.

Best CaseO(n)
Worst CaseO(n²)
Space ComplexityO(1)

Loop Invariant

At the start of each iteration of the outer loop (with index i), the subarray arr[0...i-1] consists of the elements originally in arr[0...i-1], but in sorted order.

  • Initialization: When i=1, the subarray arr[0] contains one element and is trivially sorted.
  • Maintenance: The inner while loop moves arr[j] to the left by swapping until arr[j] ≥ arr[j-1], maintaining the sorted property for arr[0...i].
  • Termination: The loop ends when i = n, meaning the subarray arr[0...n-1] (the entire array) is sorted.

1. Implementation (C++ Adjacent Swap)

void insertionSort(int arr[], int n) {
    // i is the boundary of the sorted/unsorted parts
    for (int i = 1; i < n; i++) {
        int j = i; 
        
        // Bubble the element at index j back to its correct position
        while (j > 0 && arr[j] < arr[j - 1]) {
            // Swap adjacent elements
            swap(arr[j],arr[j-1]);            
            j--; // Move pointer back
        }
    }
}

Maximum Sum Subarray

Divide & Conquer Approach | Recursive Optimization

Recurrence Relation: T(n) = T(n/2) + O(n)

Problem Statement

Given an array of integers (positive + negative), find the contiguous subarray with the maximum sum.

Example Array (A):
[ -2, 1, -3, 4, -1, 2, 1, -5, 4 ]
Indices: 0 to 8
Result: Maximum sum = 6 (Subarray: [4, -1, 2, 1])
Time ComplexityO(n log n)
Space ComplexityO(log n)
Recurrence2T(n/2) + O(n)

Divide & Conquer Strategy

For any subarray A[l..r], the maximum subarray must exist in one of three locations:

  • Left: Entirely in the left half.
  • Right: Entirely in the right half.
  • Crossing: Passing through the middle point.

We recursively compute all three and select the maximum.

1. Step-by-Step Numeric Dry Run

Step 1: Divide
Array(0..8), Mid = 4.
Left = [-2, 1, -3, 4, -1] | Right = [2, 1, -5, 4]
Step 2 & 3: Recursive Maxima
• Max in Left Half = 4
• Max in Right Half = 4
Step 4: Crossing Sum Calculation (Mid = 4)
• Left Side (4 → 0): Sums: -1, 3, 0, 1, -1. (Max Left = 3)
• Right Side (5 → 8): Sums: 2, 3, -2, 2. (Max Right = 3)
• Crossing Sum: 3 + 3 = 6
Step 5: Final Comparison
max(Left: 4, Right: 4, Crossing: 6) = 6

2. C++ Implementation

int maxCrossingSum(vector<int>& a, int l, int m, int r) {
    int sum = 0, leftSum = INT_MIN;
    for (int i = m; i >= l; i--) {
        sum += a[i];
        leftSum = max(leftSum, sum);
    }
    sum = 0;
    int rightSum = INT_MIN;
    for (int i = m + 1; i <= r; i++) {
        sum += a[i];
        rightSum = max(rightSum, sum);
    }
    return leftSum + rightSum;
}

int maxSubArray(vector<int>& a, int l, int r) {
    if (l == r) return a[l];
    int m = (l + r) / 2;
    return max({
        maxSubArray(a, l, m),
        maxSubArray(a, m + 1, r),
        maxCrossingSum(a, l, m, r)
    });
}

Intuition

Binary search is a highly efficient algorithm for finding an item from a sorted list of items. It works by repeatedly dividing in half the portion of the list that could contain the item, until you've narrowed down the possible locations to just one.

Best CaseO(1)
Worst CaseO(log n)
Space ComplexityO(log n) Recursive

In each step, we perform one comparison and then reduce the problem size to exactly half of the current range.

Merge Sort

Divide & Conquer | Stable O(n log n) Sorting

Intuition

Merge sort is a recursive algorithm that continually splits an array in half until each subarray contains a single element. It then merges those sorted subarrays back together to produce the final sorted array.

Best/Worst CaseO(n log n)
Space ComplexityO(n)
Stable?Yes

Recurrence Relation

The time complexity is governed by the two recursive calls and the linear time merge process.

T(n) = 2T(n/2) + O(n)

  • 2T(n/2): Sorting the left and right halves recursively.
  • O(n): The merge function, which compares and combines elements in linear time.
  • Result: According to Master Theorem Case 2, this yields O(n log n).

1. Loop Invariant for the Merge Step

The Merge function is the core of this algorithm. Its loop invariant ensures correctness:

  • Invariant: At the start of each iteration of the loop, the subarray arr[p..k-1] contains the k-p smallest elements of L[1..n1] and R[1..n2], in sorted order.
  • Maintenance: Comparing L[i] and R[j] and picking the smaller ensures the next smallest element is added to the main array.

• Split 1: [38, 27, 43] and [3, 9, 82, 10]
• Split 2: [38] [27, 43] | [3, 9] [82, 10]
• Split 3: [38] [27] [43] [3] [9] [82] [10] (Single elements)

• Merge [27] and [43] → [27, 43]
• Merge [38] and [27, 43] → [27, 38, 43]

void merge(int arr[], int l, int m, int r) {
    int i = l;
    int j = m + 1;
    int k = 0;

    int temp[r - l + 1];

    while (i <= m && j <= r) {
        if (arr[i] <= arr[j])
            temp[k++] = arr[i++];
        else
            temp[k++] = arr[j++];
    }

    while (i <= m)
        temp[k++] = arr[i++];

    while (j <= r)
        temp[k++] = arr[j++];

    for (int p = 0; p < k; p++)
        arr[l + p] = temp[p];
}

void mergeSort(int arr[], int l, int r) {
    if (l < r) {
        int m = l + (r - l) / 2;
        mergeSort(arr, l, m);
        mergeSort(arr, m + 1, r);
        merge(arr, l, m, r);
    }
}

Quick Sort

Divide & Conquer | First Element Pivot Strategy

Intuition

Quick Sort is a highly efficient sorting algorithm that uses a "pivot" to partition the array into two halves. This version specifically uses the first element as the pivot.

Best CaseO(n log n)
Worst CaseO(n²)
SpaceO(log n)

Exam-Critical Swap Condition

In this implementation, swapping between pointers i and j occurs ONLY when:

arr[i] > pivot AND arr[j] < pivot

  • arr[i] > pivot: Element is wrongly placed on the left side.
  • arr[j] < pivot: Element is wrongly placed on the right side.
  • The Swap: Fixes both violations simultaneously by exchanging them.

1. Implementation (C++ using Start & End)

int partition(int arr[], int start, int end) {
    int pivot = arr[start]; // First element as pivot
    int i = start + 1;
    int j = end;

    while (i <= j) {
        // Move i until an element > pivot is found
        while (i <= end && arr[i] <= pivot)
            i++;

        // Move j until an element < pivot is found
        while (arr[j] > pivot)
            j--;

        // Swap ONLY when both pointers have found violations
        if (i < j && arr[i] > pivot && arr[j] < pivot) {
            swap(arr[i], arr[j]);
        }
    }

    // Place pivot at its final sorted position
    swap(arr[start], arr[j]);
    return j;
}

void quickSort(int arr[], int start, int end) {
    if (start < end) {
        int p = partition(arr, start, end);
        quickSort(arr, start, p - 1);
        quickSort(arr, p + 1, end);
    }
}

• j starts at index 4 (value 2). 2 ≤ 8, so j stops immediately.
• i has crossed j. Final step: swap(arr[start], arr[j]).
Result of Partition:
• Array becomes: [2, 3, 6, 4, 8]
• Pivot 8 is now correctly placed at the end.

  • Best/Average Case: O(n log n). Occurs when the pivot consistently divides the array into two nearly equal halves.
  • Worst Case: O(n²). Occurs if the array is already sorted (or reverse sorted) and we always pick the first element as the pivot.

Counting Sort

Non-Comparison Sorting | Integer Range Optimization

Intuition

Counting Sort is a non-comparison based sorting algorithm. It works by counting the number of objects having distinct key values and using arithmetic to calculate the position of each object in the output sequence. It is exceptionally fast when the range of input values (k) is not significantly larger than the number of elements (n).

Time ComplexityO(n + k)
Space ComplexityO(n + k)
Stable?Yes

Algorithm Steps

  1. Find the maximum element to determine the range.
  2. Create a Count Array of size max + 1 and initialize with 0.
  3. Store the count of each element at its corresponding index.
  4. Modify the count array by adding the previous counts (Cumulative/Prefix Sum).
  5. Build the output array by placing elements at positions indicated by the count array.

1. Implementation (C++ Stable Version)

void countingSort(int arr[], int n) {
    int maxVal = arr[0];
    for (int i = 1; i < n; i++)
        if (arr[i] > maxVal) maxVal = arr[i];

    int count[maxVal + 1] = {0};

    // Step 1: Count occurrences
    for (int i = 0; i < n; i++)
        count[arr[i]]++;

    // Step 2: Cumulative count (Prefix Sum)
    for (int i = 1; i <= maxVal; i++)
        count[i] += count[i - 1];

    int output[n];

    // Step 3: Build output array (Iterate backwards for Stability)
    for (int i = n - 1; i >= 0; i--) {
        output[count[arr[i]] - 1] = arr[i];
        count[arr[i]]--;
    }

    // Step 4: Copy back to original array
    for (int i = 0; i < n; i++)
        arr[i] = output[i];
}

Bucket Sort

Distribution Sort | Optimized for Uniformly Distributed Data

Intuition

Bucket Sort is a distribution sort that works by partitioning an array into a number of buckets. Each bucket is then sorted individually, either using a different sorting algorithm or by recursively applying the bucket sorting algorithm.

Best CaseO(n + k)
Worst CaseO(n²)
Space ComplexityO(n + k)

Algorithm Steps (Scatter-Gather)

  1. Create: Initialize n empty buckets (typically as vectors or linked lists).
  2. Scatter: Put each element x into bucket at index ⌊n * x⌋.
  3. Sort: Sort each non-empty bucket individually (usually using Insertion Sort).
  4. Gather: Concatenate all sorted buckets back into the original array.

1. Implementation (C++ for [0, 1) Range)

#include <vector>
#include <algorithm>

void bucketSort(float arr[], int n) {
    // 1. Create n empty buckets
    vector<float> buckets[n];

    // 2. Scatter: Put elements in appropriate buckets
    for (int i = 0; i < n; i++) {
        int index = n * arr[i]; // Mapping [0,1) to [0,n-1]
        buckets[index].push_back(arr[i]);
    }

    // 3. Sort individual buckets
    for (int i = 0; i < n; i++)
        sort(buckets[i].begin(), buckets[i].end());

    // 4. Gather: Concatenate all buckets
    int idx = 0;
    for (int i = 0; i < n; i++)
        for (float x : buckets[i])
            arr[idx++] = x;
}

void countingSortForRadix(int arr[], int n, int exp) {
    int output[n];
    int count[10] = {0};

    // 1. Count occurrences of digits at the current 'exp' place
    for (int i = 0; i < n; i++)
        count[(arr[i] / exp) % 10]++;

    // 2. Cumulative count for stability
    for (int i = 1; i < 10; i++)
        count[i] += count[i - 1];

    // 3. Build output array (Reverse iteration ensures stability)
    for (int i = n - 1; i >= 0; i--) {
        int digit = (arr[i] / exp) % 10;
        output[count[digit] - 1] = arr[i];
        count[digit]--;
    }

    for (int i = 0; i < n; i++) arr[i] = output[i];
}

    // Counting Sort for every digit (exp is 10^i)
    for (int exp = 1; maxVal / exp > 0; exp *= 10)
        countingSortForRadix(arr, n, exp);
}

Given n points in a 2D plane, find the minimum Euclidean distance between any two points. While a brute-force approach takes $O(n^2)$, the Divide & Conquer method reduces this to $O(n \log n)$.

Distance Formula:
dist(p_1, p_2) = \sqrt{(x_1 - x_2)^2 + (y_1 - y_2)^2}
Brute ForceO(n²)
Divide & ConquerO(n log n)
Space ComplexityO(n)

Key Strategy

  1. Sort: Sort points by x-coordinate.
  2. Divide: Split points into Left and Right halves at mid.
  3. Conquer: Recursively find min distances d_L and d_R. Let d = \min(d_L, d_R).
  4. Combine (The Strip): Check points near the dividing line within a vertical strip of width $2d$. To maintain $O(n \log n)$, sort the strip by y-coordinate.

1. C++ Implementation

struct Point { int x, y; };

double dist(Point p1, Point p2) {
    return sqrt((p1.x - p2.x)*(p1.x - p2.x) + (p1.y - p2.y)*(p1.y - p2.y));
}

double stripClosest(vector<Point>& strip, double d) {
    double minDist = d;
    sort(strip.begin(), strip.end(), [](Point a, Point b) { return a.y < b.y; });

    for (int i = 0; i < strip.size(); i++)
        for (int j = i + 1; j < strip.size() && (strip[j].y - strip[i].y) < minDist; j++)
            minDist = min(minDist, dist(strip[i], strip[j]));
    return minDist;
}

double closestUtil(vector<Point>& P, int l, int r) {
    if (r - l <= 3) return bruteForce(P, l, r);

    int mid = (l + r) / 2;
    Point midPoint = P[mid];

    double d = min(closestUtil(P, l, mid), closestUtil(P, mid + 1, r));

    vector<Point> strip;
    for (int i = l; i <= r; i++)
        if (abs(P[i].x - midPoint.x) < d) strip.push_back(P[i]);

    return min(d, stripClosest(strip, d));
}

Exam-Ready Conclusion: The closest pair of points problem is solved in $O(n \log n)$ by recursively dividing points and checking a small vertical strip of width $2d$ around the midline. In the strip, each point only needs to be compared with a constant number of subsequent points (at most 7).

Formal Loop Correctness

Preconditions, Postconditions, and Hoare-Style Proofs

1. The Four Pillars of Correctness

To formally prove an algorithm's correctness, we use a framework that tracks the state of data before, during, and after a loop.

Precondition (P)True before loop starts
Loop Invariant (I)True before/after every iteration
Loop Condition (B)Controls loop execution
Postcondition (Q)True after loop terminates

Formal Proof Logic

A loop is considered correct if it satisfies the following Hoare Triple: {P} while (B) { Body } {Q}

  • Initialization ($P \Rightarrow I$): The precondition guarantees the invariant holds initially.
  • Maintenance ($\{I \land B\} \text{ Body } \{I\}$): If the invariant and condition are true, the body keeps the invariant true.
  • Termination ($I \land \neg B \Rightarrow Q$): When the loop stops, the invariant combined with the failure of the loop condition proves the postcondition.

2. Case Study: Linear Search (Formal Proof)

int i = 0;
while (i < n) {
    if (A[i] == x) return i;
    i++;
}
return -1;

Precondition (P): Array A[0...n-1] exists and x is the target.

Loop Invariant (I): At the start of each iteration, x is not present in A[0...i-1].

Postcondition (Q): If loop ends, x is not present in the entire array A[0...n-1].

Initialization: i=0. Subarray A[0...-1] is empty. Invariant holds trivially.

Maintenance: If A[i] != x, we increment i. The element was not at i, and by invariant, not before i, so it is not in A[0...i].

Termination: Loop ends when i = n. Combined with Invariant, x is not in A[0...n-1].

3. Algorithm Invariant Gallery

Algorithm Loop Invariant (at start of iteration i)
Insertion Sort Subarray A[0...i-1] is sorted.
Bubble Sort The last i elements are in their final sorted positions.
Binary Search If x exists, it is within A[low...high].
Selection Sort A[0...i-1] contains the i smallest elements in sorted order.

4. Exam Checklist for Correctness Proofs

  • Be Specific: Use array indices (e.g., A[0...i-1]) rather than "the first part".
  • Timing: Always state "At the start of each iteration...".
  • Maintenance logic: Explicitly show how the loop body preserves the property.
  • Final Logic: Show how the invariant leads directly to the sorted or searched result.
One-Line Exam Gold: A loop invariant is a property that holds before and after each iteration of a loop and is used to prove algorithm correctness via Initialization, Maintenance, and Termination.

Heap Sort

Comparison Sort | Complete Binary Tree Representation

Intuition

Heap Sort utilizes a Max Heap data structure to repeatedly identify and "extract" the maximum element. By moving the root (max) to the end of the array and restoring the heap property, the array is sorted in O(n log n) time.

Build HeapO(n)
ExtractionO(n log n)
Space ComplexityO(n) with ans vector

The Two-Step Process

  1. Build Max Heap: Convert the unsorted array into a Max Heap where the parent is always greater than its children.
  2. Extract Max: Repeatedly take the root element (maximum), add it to your results, swap it with the last leaf, and perform Heapify to maintain the heap property.

1. Implementation (C++ with Extraction Vector)

void heapify(vector<int>& A, int n, int i) {
    int largest = i;
    int left = 2 * i + 1;
    int right = 2 * i + 2;

    if (left < n && A[left] > A[largest]) largest = left;
    if (right < n && A[right] > A[largest]) largest = right;

    if (largest != i) {
        swap(A[i], A[largest]);
        heapify(A, n, largest);
    }
}

vector<int> heapSortExtract(vector<int>& A) {
    int n = A.size();
    vector<int> ans;

    // Step 1: Build Max Heap
    for (int i = n / 2 - 1; i >= 0; i--)
        heapify(A, n, i);

    // Step 2: Extract elements into ans vector
    for (int i = n - 1; i >= 0; i--) {
        ans.push_back(A[0]); // Extract max (Root)
        swap(A[0], A[i]);    // Move current root to end
        heapify(A, i, 0);    // Restore heap on reduced size
    }
    return ans; // Note: This returns Descending order
}

3. Properties & Performance

  • Stability: No. Relative order of equal keys is not preserved during swaps.
  • In-place: Usually yes (if sorting within the original array), but this version uses O(n) space due to the ans vector.
  • Worst/Best Case: Always O(n log n) due to the height of the complete binary tree.
Exam Insight: Building a heap takes O(n) time, not O(n log n). The log n factor only comes into play during the n-1 extraction steps, where each extraction requires a heapify operation of log n depth.
